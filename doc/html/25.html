输出栏目导航

使用以下代码可以方便的输出栏目列表 

{{n()}}


可以使用以下代码获取导航列表数组后输出 

{{$set=array()}}

{{$set['cid']=`cid}}

{{$set['returnarray']=1}}

{{$nav=n($set)}}

{{loop $nav as $a}}

<li {{$a['onstyle']}}><a href="{{$a['url']}}">{{$a['name']}}</a></li>

{{/loop}}

n()函数支持传入数组设置

数组参数如下 

<?php
unset($set);
$set['cid']=cid;//栏目也左侧显示下属栏目
$set['cid']=[1,2,3];//以数组方式传入cid,列出这些栏目
$set['onstyle']='style="color:#FF0000"';
$set['bro']=1;//默认如果无下属栏目,则显示兄弟栏目
$set['papa']=1;//默认显示父栏目
$set['returnarray']=1;//以数组的形式返回栏目列表,方便手动循环输出栏目
//新定义首页名字链接,不想显示首页则$set['homepage']='';
$set['homepage']['name']="首页" ;
$set['homepage']['url']="/";
echo(n($set));
?>
